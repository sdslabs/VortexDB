# SDK Examples

Ready-to-run code examples demonstrating VortexDB Python SDK usage.

## Basic Usage

```python
from vortexdb import (
    VortexDB, DenseQuery, DenseVector, Payload, PointInput, SearchQuery, Similarity, SparseVector,
)

with VortexDB(grpc_url="localhost:50051", api_key="secret") as db:
    # Insert
    point_id = db.insert(
        dense_vector=DenseVector([0.1, 0.2, 0.3]),
        payload=Payload.text("hello world"),
    )
    print(f"Inserted: {point_id}")

    # Batch insert: dense, sparse, or both on one point
    ids = db.batch_insert(points=[
        PointInput(dense_vector=DenseVector([0.1, 0.2, 0.3]), payload=Payload.text("doc one")),
        PointInput(sparse_vector=SparseVector([4, 8], [1.0, 0.5]), payload=Payload.text("doc two")),
    ])

    # Dense search: score is a distance, lower is closer
    hits = db.search(query=SearchQuery(3, DenseQuery(DenseVector([0.1, 0.2, 0.3]), Similarity.COSINE)))
    print(f"Found {len(hits)} results")

    # Sparse search: score is a dot product, higher is closer
    hits = db.search(query=SearchQuery(3, SparseVector([8], [1.0])))
```

## Semantic Search

Using sentence-transformers for text embedding:

```python
from vortexdb import VortexDB, DenseQuery, DenseVector, Payload, SearchQuery, Similarity
from sentence_transformers import SentenceTransformer

model = SentenceTransformer('all-MiniLM-L6-v2')

documents = [
    "The quick brown fox jumps over the lazy dog",
    "Machine learning is a subset of artificial intelligence",
    "Python is a popular programming language",
]

def embed(text: str) -> DenseVector:
    return DenseVector(model.encode(text).tolist())

with VortexDB(grpc_url="localhost:50051", api_key="secret") as db:
    for doc in documents:
        db.insert(dense_vector=embed(doc), payload=Payload.text(doc))

    hits = db.search(query=SearchQuery(
        limit=2,
        query=DenseQuery(embed("AI and programming"), Similarity.COSINE),
    ))
    for hit in hits:
        point = db.get(point_id=hit.id)
        print(f"  {point.payload.content}")
```

## Batch Processing

```python
from vortexdb import (
    VortexDB, DenseQuery, DenseVector, Payload, PointInput, SearchQuery, Similarity, SparseVector,
)

with VortexDB(grpc_url="localhost:50051", api_key="secret") as db:
    points = [
        PointInput(dense_vector=DenseVector([(i + 1) * 0.1] * 3), payload=Payload.text(f"doc {i}"))
        for i in range(100)
    ]
    ids = db.batch_insert(points=points)

    # Batch search, dense and sparse queries can be mixed
    queries = [
        SearchQuery(3, DenseQuery(DenseVector([0.1, 0.2, 0.3]), Similarity.COSINE)),
        SearchQuery(3, DenseQuery(DenseVector([0.4, 0.5, 0.6]), Similarity.EUCLIDEAN)),
        SearchQuery(3, SparseVector([1, 2], [1.0, 1.0])),
    ]
    batch_results = db.batch_search(queries=queries)
    for i, res in enumerate(batch_results):
        print(f"Query {i}: {len(res)} results")
```

## Testing with pytest

```python
import pytest
from vortexdb import VortexDB, DenseQuery, DenseVector, Payload, SearchQuery, Similarity

@pytest.fixture
def db():
    client = VortexDB(grpc_url="localhost:50051", api_key="secret")
    yield client
    client.close()

class TestVortexDB:
    def test_insert_and_get(self, db):
        point_id = db.insert(
            dense_vector=DenseVector([0.1, 0.2, 0.3, 0.4]),
            payload=Payload.text("Test document"),
        )
        point = db.get(point_id=point_id)
        assert point is not None
        assert point.payload.content == "Test document"
        db.delete(point_id=point_id)

    def test_search(self, db):
        point_id = db.insert(
            dense_vector=DenseVector([1.0, 2.0, 3.0, 4.0]),
            payload=Payload.text("target"),
        )
        hits = db.search(query=SearchQuery(
            limit=10,
            query=DenseQuery(DenseVector([1.0, 2.0, 3.0, 4.0]), Similarity.COSINE),
        ))
        assert point_id in [hit.id for hit in hits]
        db.delete(point_id=point_id)
```
